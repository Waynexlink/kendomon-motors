Good timing to think this through carefully. Here's the full picture before we write any code:

Why CAPI matters for you specifically: Your audience is in Nigeria, many on mobile browsers that block cookies or use ad blockers. The browser pixel alone will miss a significant chunk of conversions. CAPI fires server-side — it bypasses all of that.

The deduplication problem: If both the browser pixel AND CAPI fire a Lead event for the same click, Meta counts it twice and your cost-per-result looks artificially low, throwing off optimization. The fix is a shared event_id — you generate one ID on the frontend, pass it to both fbq() and your CAPI call, and Meta ignores the duplicate.

The full data flow:

 
User clicks "Continue to WhatsApp"
│
├──► fbq('track', 'Lead', data, { eventID: 'abc-123' }) ← browser pixel
│
└──► POST /netlify/functions/meta-capi { event_id: 'abc-123', ... }
│
└──► Meta Graph API (server-side)

Meta sees same event_id on both → counts as ONE conversion ✓

2. Netlify Function for CAPI

Create this file at netlify/functions/meta-capi.js — Netlify auto-detects anything in that folder:

 
js
const crypto = require("crypto");

function sha256(value) {
if (!value) return null;
return crypto
.createHash("sha256")
.update(value.trim().toLowerCase())
.digest("hex");
}

exports.handler = async (event) => {
// Only accept POST
if (event.httpMethod !== "POST") {
return { statusCode: 405, body: "Method Not Allowed" };
}

let body;
try {
body = JSON.parse(event.body);
} catch {
return { statusCode: 400, body: "Invalid JSON" };
}

const {
event_id,
event_name = "Lead",
event_source_url,
fbp,
fbc,
client_user_agent,
content_name,
} = body;

// Get real IP from Netlify headers
const client_ip_address =
event.headers["x-forwarded-for"]?.split(",")[0]?.trim() ||
event.headers["client-ip"] ||
null;

const pixelId = process.env.META_PIXEL_ID;
const accessToken = process.env.META_ACCESS_TOKEN;

if (!pixelId || !accessToken) {
console.error("[meta-capi] Missing env vars");
return { statusCode: 500, body: "Server configuration error" };
}

const payload = {
data: [
{
event_name,
event_time: Math.floor(Date.now() / 1000),
event_id,
event_source_url,
action_source: "website",
user_data: {
// We don't collect email/phone in the modal so these are null
// Once you add a contact form later, hash and send em, fbn here
em: null,
ph: null,
client_ip_address,
client_user_agent,
// fbp and fbc come from the browser pixel cookies
// they help Meta match the event to a user account
fbp: fbp || null,
fbc: fbc || null,
},
custom_data: {
content_name: content_name || "",
content_category: "Car Enquiry",
},
},
],
// test_event_code: "TEST12345", // ← uncomment this during testing only
};

try {
const res = await fetch(
`https://graph.facebook.com/v19.0/${pixelId}/events?access_token=${accessToken}`,
{
method: "POST",
headers: { "Content-Type": "application/json" },
body: JSON.stringify(payload),
}
);

const result = await res.json();

if (!res.ok) {
console.error("[meta-capi] Meta API error:", result);
return {
statusCode: 500,
body: JSON.stringify({ error: result }),
};
}

console.log("[meta-capi] Event sent:", result);
return {
statusCode: 200,
body: JSON.stringify({ success: true, events_received: result.events_received }),
};
} catch (err) {
console.error("[meta-capi] Fetch error:", err);
return { statusCode: 500, body: "Internal error" };
}
};

Step 2 — Creating the Netlify Function file.


What is a Netlify Function?
Your site is static — it is just HTML, CSS and JavaScript files. It cannot talk to Meta's servers directly from the browser safely because that would expose your secret token to anyone who looks at your page source.

A Netlify Function is a small piece of server code that Netlify runs for you in the background. Your site sends data to it privately, it adds your secret token, and forwards everything to Meta. Nobody visiting your site can see any of it.

js
import fs from "fs";
import path from "path";

const textFilePath = "./car-data.txt";
const targetDirectory = "./src/content/cars";

if (!fs.existsSync(targetDirectory)) {
fs.mkdirSync(targetDirectory, { recursive: true });
}

function slugify(text) {
return text
.toString()
.toLowerCase()
.replace(/\s+/g, "-")
.replace(/[^\w\-]+/g, "")
.replace(/\-\-+/g, "-")
.trim("-");
}

const rawData = fs.readFileSync(textFilePath, "utf8");
const carBlocks = rawData.split("===").map(b => b.trim()).filter(Boolean);

carBlocks.forEach((block) => {
const car = {};

block.split("\n").forEach(line => {
const colonIndex = line.indexOf(":");
if (colonIndex === -1) return;
const key = line.slice(0, colonIndex).trim();
const value = line.slice(colonIndex + 1).trim();
car[key] = value;
});

if (!car.make || !car.model || !car.year || !car.price) {
console.log(`Skipping — missing required fields:`, car);
return;
}

const slug = slugify(`${car.year}-${car.make}-${car.model}`);
const filePath = path.join(targetDirectory, `${slug}.md`);

if (fs.existsSync(filePath)) {
console.log(`Already exists, skipping: ${slug}.md`);
return;
}

// Gallery images as proper YAML list (this is what Astro needs)
let gallerySection = "";
if (car.galleryImages) {
const images = car.galleryImages
.split(",")
.map(img => img.trim())
.filter(Boolean);

if (images.length > 0) {
gallerySection = `\ngalleryImages:\n${images.map(img => ` - ./${img}`).join("\n")}`;
}
}

const markdown = `---
make: ${car.make}
model: ${car.model}
year: ${parseInt(car.year)}
price: ${parseFloat(car.price)}
transmission: ${car.transmission || "Automatic"}
condition: ${car.condition || "Foreign Used"}
location: "${car.location || "Gwarinpa, Abuja"}"
description: "${car.description || ""}"
heroImage: ./${car.heroImage}
heroImageAlt: "${car.year} ${car.make} ${car.model} for sale in Abuja"${gallerySection}
featured: false
publishDate: ${new Date().toISOString().split("T")[0]}
---

${car.description || `${car.year} ${car.make} ${car.model} available for inspection and purchase at our Gwarinpa showroom.`}
`;

fs.writeFileSync(filePath, markdown, "utf8");
console.log(`✅ Created: ${slug}.md`);
});

console.log("\nDone.");
